! =====================================================================
! DISTRIBUTION switch template  (D-SW1 .. D-SW6)  — Layer 2
! Role: aggregate a faculty's access switches; dual-home to BOTH cores
!       (STP blocks the redundant uplink = loop-free redundancy).
! Rename hostname per switch (D-SW1, D-SW2, ...).
! =====================================================================
enable
configure terminal
hostname D-SW1
no ip domain-lookup
!
username admin privilege 15 secret Str0ng-Admin!
enable secret Str0ng-Enable!
ip domain-name univ.local
service password-encryption
banner motd #Authorized access only.#
!
spanning-tree mode rapid-pvst
!
! ---- VLANs this switch must carry (define locally, or use VTP) ----
vlan 20
 name SCIENCE        ! change to this distribution block's faculty VLAN
vlan 99
 name MGMT
exit
!
! ---- Uplinks to the two cores (STP keeps one forwarding, one blocking) ----
interface gigabitEthernet0/1
 description === Uplink to CORE-SW1 ===
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 20,99
 no shutdown
interface gigabitEthernet0/2
 description === Uplink to CORE-SW2 ===
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 20,99
 no shutdown
exit
!
! ---- Downlinks (trunks) to the faculty access switches ----
interface range fastEthernet0/1 - 10
 description === Trunk to Access switches ===
 switchport mode trunk
 switchport trunk allowed vlan 20,99
 no shutdown
exit
!
! ---- shut unused ports ----
interface range fastEthernet0/11 - 24
 shutdown
exit
!
! ---- management access (VLAN 99 SVI) ----
interface vlan 99
 ip address 192.168.99.11 255.255.255.0     ! .11 for D-SW1, .12 for D-SW2, ...
 no shutdown
exit
ip default-gateway 192.168.99.1
!
crypto key generate rsa modulus 2048
ip ssh version 2
line vty 0 4
 transport input ssh
 login local
 exec-timeout 5 0
exit
end
write memory
!
! Repeat for D-SW2..6, changing: hostname, the faculty VLAN (30/40/50/60/70),
! the 'allowed vlan' lists, and the VLAN 99 management IP.
! VERIFY: show spanning-tree  (one uplink FWD, one BLK);  show interfaces trunk
