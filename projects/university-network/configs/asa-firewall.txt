! =====================================================================
! ASA — Edge Firewall (ASA 5506/5505 in Packet Tracer)
! Role: stateful firewall between the campus (inside) and the edge (outside).
!   inside  (security 100) = to CORE  192.168.200.1/29
!   outside (security 0)   = to R1    203.0.113.2/30
! Stateful rule of thumb: high->low allowed by default; low->high needs an ACL.
! =====================================================================
enable
configure terminal
hostname ASA
domain-name univ.local
!
! ---- interfaces ----
interface gigabitEthernet1/1
 nameif outside
 security-level 0
 ip address 203.0.113.2 255.255.255.252
 no shutdown
!
interface gigabitEthernet1/2
 nameif inside
 security-level 100
 ip address 192.168.200.1 255.255.255.248
 no shutdown
exit
!
! ---- routing ----
route outside 0.0.0.0 0.0.0.0 203.0.113.1 1          ! default to R1 / Internet
route inside 192.168.0.0 255.255.0.0 192.168.200.2 1  ! campus via the core
!
! ---- allow inside users out (inside is higher security, but be explicit) ----
access-list INSIDE-OUT extended permit ip 192.168.0.0 255.255.0.0 any
access-group INSIDE-OUT in interface inside
!
! ---- from the Internet, only allow what you publish (example: web server) ----
! (uncomment + add a static NAT if you expose a public server)
! access-list OUTSIDE-IN extended permit tcp any host 203.0.113.2 eq 80
! access-group OUTSIDE-IN in interface outside
!
! ---- stateful inspection (return traffic is allowed automatically) ----
class-map inspection_default
 match default-inspection-traffic
policy-map global_policy
 class inspection_default
  inspect icmp
service-policy global_policy global
!
! ---- management ----
username admin password Str0ng-Admin! privilege 15
aaa authentication ssh console LOCAL
crypto key generate rsa modulus 2048
ssh 192.168.99.0 255.255.255.0 inside
ssh 192.168.10.0 255.255.255.0 inside
ssh version 2
ssh timeout 10
!
end
write memory
!
! NOTES
! - In ASA 8.3+ NAT is not required to pass traffic (nat-control is off), so the
!   single campus NAT is done on R1. If your PT ASA version needs NAT, add:
!     object network CAMPUS
!      subnet 192.168.0.0 255.255.0.0
!      nat (inside,outside) dynamic interface
! - VERIFY: show interface ip brief ; show route ; show conn ; show access-list
